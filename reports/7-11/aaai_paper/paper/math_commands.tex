%% ---- cross-document reference macros -------------------------------------
%% \appref{label}{letter}  : main text -> appendix
%% \mainref{label}{number} : appendix -> main text section
%% \mainfig{label}{number} : appendix -> main text figure
%% \maintab{label}{number} : appendix -> main text table
%% When \ifappendixinline is true the appendix ships inside main.tex and these
%% resolve via \ref; otherwise they degrade to literal cross-PDF wording.
\ifappendixinline
  \newcommand{\appref}[2]{Appendix~\ref{#1}}
  \newcommand{\mainref}[2]{\S\ref{#1}}
  \newcommand{\mainfig}[2]{Figure~\ref{#1}}
  \newcommand{\maintab}[2]{Table~\ref{#1}}
\else
  %% No parentheses here: every call site is already wrapped in "(...)",
  %% and "(Appendix B (supplementary))" nests badly.
  \newcommand{\appref}[2]{Appendix~#2 of the supplement}
  \newcommand{\mainref}[2]{\S#2 of the main paper}
  \newcommand{\mainfig}[2]{Figure~#2 of the main paper}
  \newcommand{\maintab}[2]{Table~#2 of the main paper}
\fi
